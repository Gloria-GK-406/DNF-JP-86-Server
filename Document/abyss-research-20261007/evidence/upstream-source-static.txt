57:         {
58:             var rules = DifficultyRules.Value;
59:             rules.TryGetValue('A', out var veryHard);
60:             rules.TryGetValue('B', out var hard);
61:
62:             // PVF [difficulty] item 4 is the manual A/B selection weight.
63:             // Keep the random result in the selection owner so the chosen
64:             // HellPartyRoomInfo is frozen once and reused by all projections.
65:             var veryHardWeight = Math.Max(0, veryHard?.Probability ?? 0);
66:             var hardWeight = Math.Max(0, hard?.Probability ?? 0);
67:             var total = veryHardWeight + hardWeight;
68:             if (total <= 0)
69:                 return 1;
70:
71:             var roll = Infrastructure.ServerRandom.Next(total);
72:             return roll < veryHardWeight ? (byte)1 : (byte)2;

110:         private static void ParseSequentialGroupNodes(
111:             IReadOnlyList<ScriptNode> nodes,
112:             string content,
113:             Dictionary<int, HellPartyMonsterGroup> result)
114:         {
115:             HellPartyMonsterGroup current = null;
116:             for (var i = 0; i < nodes.Count; i++)
117:             {
118:                 var node = nodes[i];
119:                 if (node.Tag.Equals("group index", StringComparison.OrdinalIgnoreCase))
120:                 {
121:                     var groupIndex = ParseFirstInt(node.GetFirstDataContent(content));
122:                     if (groupIndex < 0)
123:                     {
124:                         current = null;
125:                         continue;
126:                     }
127:
128:                     if (!result.TryGetValue(groupIndex, out current))
129:                     {
130:                         current = new HellPartyMonsterGroup();
131:                         result[groupIndex] = current;
132:                     }
133:                     continue;
134:                 }
135:
136:                 if (current != null && node.Tag.Equals("group", StringComparison.OrdinalIgnoreCase))
137:                     ParseGroupNode(node, content, current);
138:             }
139:         }

184:         private static void ParseGroupNode(ScriptNode node, string content, HellPartyMonsterGroup target)
185:         {
186:             var tokens = JoinDataItems(node, content)
187:                 .Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
188:
189:             var difficulty = '\0';
190:             int? pendingCode = null;
191:             for (var i = 0; i < tokens.Length; i++)
192:             {
193:                 var token = StripBacktick(tokens[i]);
194:                 if (token.Length == 1 && char.IsLetter(token[0]))
195:                 {
196:                     difficulty = char.ToUpperInvariant(token[0]);
197:                     pendingCode = null;
198:                     target.Ensure(difficulty);
199:                     continue;
200:                 }
201:
202:                 if (!int.TryParse(token, out var value))
203:                     continue;
204:
205:                 if (difficulty == '\0')
206:                     difficulty = 'A';
207:
208:                 if (!pendingCode.HasValue)
209:                 {
210:                     pendingCode = value;
211:                     continue;
212:                 }
213:
214:                 target.Ensure(difficulty).Add(new HellPartyGroupEntry
215:                 {
216:                     Code = pendingCode.Value,
217:                     EntityType = value == 1 ? (byte)1 : (byte)0,
218:                 });
219:                 pendingCode = null;
220:             }
221:         }

223:         private static void ParseDifficultyNode(
224:             ScriptNode node,
225:             string content,
226:             Dictionary<char, HellPartyDifficultyRule> result)
227:         {
228:             var tokens = JoinDataItems(node, content)
229:                 .Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
230:
231:             var i = 0;
232:             if (i < tokens.Length && int.TryParse(StripBacktick(tokens[i]), out _))
233:                 i++;
234:
235:             while (i < tokens.Length)
236:             {
237:                 var keyToken = StripBacktick(tokens[i++]);
238:                 if (keyToken.Length != 1 || !char.IsLetter(keyToken[0]))
239:                     continue;
240:
241:                 var values = new int[5];
242:                 var valueCount = 0;
243:                 while (i < tokens.Length && valueCount < values.Length)
244:                 {
245:                     var token = StripBacktick(tokens[i++]);
246:                     if (int.TryParse(token, out var value))
247:                         values[valueCount++] = value;
248:                 }
249:
250:                 if (valueCount != values.Length)
251:                     continue;
252:
253:                 result[char.ToUpperInvariant(keyToken[0])] = new HellPartyDifficultyRule
254:                 {
255:                     Key = char.ToUpperInvariant(keyToken[0]),
256:                     RewardRollCount = values[0],
257:                     Unknown1 = values[1],
258:                     RatioProbability = values[2],
259:                     Probability = values[3],
260:                     Unknown4 = values[4],
261:                 };
262:             }
263:         }
264:
265:         private static string JoinDataItems(ScriptNode node, string content)
266:         {
267:             if (node == null || node.DataItems.Count == 0)
268:                 return string.Empty;
269:
270:             var parts = new List<string>();
271:             foreach (var item in node.DataItems)
272:                 parts.Add(item.GetContent(content).Trim());
273:             return string.Join(" ", parts);
274:         }
275:
276:         private static int ParseFirstInt(string text)
277:         {
278:             if (string.IsNullOrWhiteSpace(text))
279:                 return -1;
280:
281:             var tokens = text.Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
282:             for (var i = 0; i < tokens.Length; i++)
283:                 if (int.TryParse(StripBacktick(tokens[i]), out var value))
284:                     return value;
285:             return -1;
286:         }
287:
288:         private static string StripBacktick(string value)
289:         {
290:             if (value != null && value.Length >= 2 && value[0] == '`' && value[value.Length - 1] == '`')
291:                 return value.Substring(1, value.Length - 2);
292:             return value ?? string.Empty;
293:         }

295:         private static char HellModeToGroupKey(byte difficulty)
296:         {
297:             return difficulty == 2 ? 'B' : 'A';
298:         }

322:

324:             {
325:                 return _entries.TryGetValue(difficulty, out var exact) && exact.Count > 0;
326:             }
327:         }
328:     }

28:         public static List<DropInfo> GenerateSpecificEquipmentDrops(
29:             DnfLcg lcg,
30:             int dungeonMinimumLevel,
31:             int itemBasisLevel,
32:             int dungeonDifficulty,
33:             byte hellDifficulty,
34:             int rewardRollCount,
35:             bool epicBuffPotionActive,
36:             ref ushort slotCounter)
37:         {
38:             EnsureLoaded();
39:             var result = new List<DropInfo>();
40:             if (lcg == null || rewardRollCount <= 0)
41:                 return result;
42:
43:             var prob = FindDropProb(itemBasisLevel);
44:             if (prob.DifficultyRates == null)
45:                 return result;
46:
47:             var difficultyIndex = Math.Max(0, Math.Min(dungeonDifficulty, prob.DifficultyRates.Length - 1));
48:             var rate = prob.DifficultyRates[difficultyIndex];
49:             if (rate <= 0)
50:                 return result;
51:
52:             // 70 服务端以基础倍率乘深渊概率表；当前模拟器没有独立基础倍率来源，先固定为 100。
53:             var chance = BaseDropRate * rate / 100;
54:             for (var i = 0; i < rewardRollCount; i++)
55:             {
56:                 var chanceRoll = lcg.Next(DropChanceDenominator);
57:                 if (chanceRoll > chance)
58:                 {
59:                     FileLogger.Log($"[HellMonsterDropConfig] roll#{i + 1}: miss itemBasisLevel={itemBasisLevel} dungeonDifficulty={dungeonDifficulty} hellDifficulty={hellDifficulty} rate={rate} chance={chance} roll={chanceRoll}");
60:                     continue;
61:                 }
62:
63:                 var initialRarity = RollHellRarity(lcg, hellDifficulty);
64:                 var rarity = ApplyEpicBuffPotionRarityReroll(
65:                     initialRarity,
66:                     epicBuffPotionActive,
67:                     () => RollHellRarity(lcg, hellDifficulty),
68:                     out var rerollRarity);
69:                 var itemId = ChooseEquipment(
70:                     lcg,
71:                     dungeonMinimumLevel,
72:                     itemBasisLevel,
73:                     rarity,
74:                     allowFallback: true,
75:                     predicate: null,
76:                     out var candidateCount,
77:                     out var fallbackUsed,
78:                     out var gradeMin,
79:                     out var gradeMax);
80:                 if (itemId <= 0)
81:                 {
82:                     FileLogger.Log($"[HellMonsterDropConfig] roll#{i + 1}: no equipment dungeonLevelRange={dungeonMinimumLevel}-{itemBasisLevel} gradeRange={gradeMin}-{gradeMax} rarity={rarity} epicBuff={FormatEpicBuffReroll(epicBuffPotionActive, initialRarity, rerollRarity)} candidates={candidateCount} fallback={fallbackUsed}");
83:                     continue;
84:                 }
85:
86:                 var meta = ItemMetadataResolver.Resolve(itemId);
87:                 slotCounter++;
88:                 result.Add(DropInfo.CreateItem(slotCounter, itemId, 1));
89:                 FileLogger.Log($"[HellMonsterDropConfig] roll#{i + 1}: hit dungeonLevelRange={dungeonMinimumLevel}-{itemBasisLevel} gradeRange={gradeMin}-{gradeMax} dungeonDifficulty={dungeonDifficulty} hellDifficulty={hellDifficulty} rate={rate} chance={chance} roll={chanceRoll} rarity={rarity} epicBuff={FormatEpicBuffReroll(epicBuffPotionActive, initialRarity, rerollRarity)} candidates={candidateCount} fallback={fallbackUsed} item={itemId} itemGrade={meta.Grade} itemMinLv={meta.MinimumLevel}");
90:             }
91:
92:             return result;
93:         }

170:         private static void ParseDropProb(string text)
171:         {
172:             var vals = ParseInts(ExtractSection(text, "dungeon difficulty drop prob"));
173:             if (vals.Count < 7)
174:                 return;
175:
176:             var rows = vals.Count / 7;
177:             _dropProbs = new DropProb[rows];
178:             for (var row = 0; row < rows; row++)
179:             {
180:                 var baseIndex = row * 7;
181:                 var rates = new int[5];
182:                 for (var i = 0; i < rates.Length; i++)
183:                     rates[i] = vals[baseIndex + 2 + i];
184:
185:                 _dropProbs[row] = new DropProb
186:                 {
187:                     LevelMin = vals[baseIndex],
188:                     LevelMax = vals[baseIndex + 1],
189:                     DifficultyRates = rates,
190:                 };
191:             }
192:             FileLogger.Log($"[HellMonsterDropConfig] [dungeon difficulty drop prob]: rows={rows}");
193:         }

195:         private static void ParseRarityThresholds(string text)
196:         {
197:             var vals = ParseInts(ExtractSection(text, "basis of rarity dicision"));
198:             if (vals.Count < 2)
199:                 return;
200:
201:             var rowCount = Math.Max(0, vals[0]);
202:             if (rowCount <= 0)

206:             var index = 1;

208:             {

210:                 for (var col = 0; col < thresholds.Length && index < vals.Count; col++)
211:                     thresholds[col] = vals[index++];

213:             }

247:         private static int RollHellRarity(DnfLcg lcg, byte hellDifficulty)
248:         {
249:             EnsureLoaded();
250:             if (_rarityThresholds == null || _rarityThresholds.Length == 0)
251:                 return 0;
252:
253:             // hellDifficulty: 1=A/非常困难，2=B/困难；稀有度表数组从 0 开始。
254:             var row = Math.Max(0, Math.Min((int)hellDifficulty - 1, _rarityThresholds.Length - 1));
255:             var thresholds = _rarityThresholds[row];
256:             if (thresholds == null || thresholds.Length == 0)
257:                 return 0;
258:
259:             var roll = lcg.Next(1000000) + 1;
260:             var limit = Math.Min(6, thresholds.Length);
261:             for (var rarity = 0; rarity < limit; rarity++)
262:             {
263:                 if (roll <= thresholds[rarity])
264:                     return rarity;
265:             }
266:             return 0;
267:         }

314:         private static int ChooseEquipment(
315:             DnfLcg lcg,
316:             int dungeonMinimumLevel,
317:             int itemBasisLevel,
318:             int rarity,
319:             bool allowFallback,
320:             Func<int, bool> predicate,
321:             out int candidateCount,
322:             out bool fallbackUsed,
323:             out int gradeMin,
324:             out int gradeMax)
325:         {
326:             candidateCount = 0;
327:             fallbackUsed = false;
328:             gradeMin = 0;
329:             gradeMax = 0;
330:             EnsureLoaded();
331:             if (_equipPool == null)
332:                 return -1;
333:
334:             if (!EquipmentDropLevelRule.TryGetAtlasGradeRange(dungeonMinimumLevel, itemBasisLevel, out gradeMin, out gradeMax))
335:                 return -1;
336:             var candidates = CollectWeightedFromPool(
337:                 gradeMin,
338:                 gradeMax,
339:                 rarity,
340:                 predicate);
341:             if (candidates.Count == 0 && allowFallback && rarity > 0)
342:             {
343:                 candidates = CollectWeightedFromPool(
344:                     gradeMin,
345:                     gradeMax,
346:                     0,
347:                     predicate);
348:                 fallbackUsed = true;
349:             }
350:             candidateCount = candidates.Count;
351:             if (candidates.Count == 0)
352:                 return -1;
353:
354:             var totalWeight = 0;
355:             for (var i = 0; i < candidates.Count; i++)
356:                 totalWeight += candidates[i].Weight;
357:             if (totalWeight <= 0)
358:                 return -1;
359:
360:             var roll = lcg.Next(totalWeight);
361:             var cumulative = 0;
362:             for (var i = 0; i < candidates.Count; i++)
363:             {
364:                 cumulative += candidates[i].Weight;
365:                 if (roll < cumulative)
366:                     return candidates[i].Id;
367:             }
368:             return candidates[candidates.Count - 1].Id;
369:         }

371:         private static bool TryGetGradeRangeFromItemDropRefTable(int itemBasisLevel, out int gradeMin, out int gradeMax)
372:         {
373:             gradeMin = 0;
374:             gradeMax = 0;
375:             if (_gradeRanges == null || itemBasisLevel < 1 || itemBasisLevel > _gradeRanges.Length)
376:                 return false;
377:
378:             var range = _gradeRanges[itemBasisLevel - 1];
379:             if (range.Down <= 0 && range.Up <= 0)
380:                 return false;
381:
382:             // [item drop ref table] 按副本掉落基准等级查表，返回可抽取装备 .equ [grade] 范围。
383:             // 表内 down/up 包含基准等级本身，例如 83 7 4 => 83-6 到 83+3，即 77..86。
384:             var downOffset = Math.Max(0, range.Down - 1);
385:             var upOffset = Math.Max(0, range.Up - 1);
386:             gradeMin = Math.Max(1, itemBasisLevel - downOffset);
387:             gradeMax = Math.Min(200, itemBasisLevel + upOffset);
388:             return gradeMax >= gradeMin;
389:         }

391:         private static List<(int Id, int Weight)> CollectWeightedFromPool(
392:             int gradeMin,
393:             int gradeMax,
394:             int rarity,
395:             Func<int, bool> predicate)
396:         {
397:             var result = new List<(int Id, int Weight)>();
398:             if (gradeMax < gradeMin)
399:                 return result;
400:
401:             for (var grade = gradeMin; grade <= gradeMax; grade++)
402:             {
403:                 var key = (long)grade * 10 + rarity;
404:                 if (_equipPool.TryGetValue(key, out var items))
405:                 {
406:                     for (var i = 0; i < items.Count; i++)
407:                     {
408:                         var item = items[i];
409:                         if (predicate == null || predicate(item.Id))
410:                             result.Add(item);
411:                     }
412:                 }
413:             }
414:             return result;
415:         }

417:         private static string ExtractSection(string text, string name)
418:         {
419:             var tag = "[" + name + "]";
420:             var start = text.IndexOf(tag, StringComparison.OrdinalIgnoreCase);
421:             if (start < 0)
422:                 return null;
423:
424:             start += tag.Length;
425:             var end = text.IndexOf("[", start, StringComparison.Ordinal);
426:             if (end < 0)
427:                 end = text.Length;
428:             return text.Substring(start, end - start);
429:         }
430:
431:         private static List<int> ParseInts(string section)
432:         {
433:             var result = new List<int>();
434:             if (string.IsNullOrWhiteSpace(section))
435:                 return result;
436:
437:             var tokens = section.Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
438:             for (var i = 0; i < tokens.Length; i++)
439:             {
440:                 if (int.TryParse(tokens[i], out var value))
441:                     result.Add(value);
442:             }
443:             return result;
444:         }

21:                 dungeonMinimumLevel = dungeonBasisLevel;
22:                 dungeonBasisLevel = tmp;
23:             }
24:
25:             // 装备图鉴展示的可掉落副本等级为: 装备 grade - 7 到装备 grade。
26:             // 副本自身按 [minimum required level, basis level] 作为等级区间，两段相交即可掉落。
27:             gradeMin = Math.Max(1, dungeonMinimumLevel);
28:             gradeMax = Math.Min(200, dungeonBasisLevel + 7);
29:             return gradeMax >= gradeMin;
30:         }
31:     }

121:             var recipeBlock = ExtractTaggedBlock(text, "equipment recipe info");
122:             if (string.IsNullOrWhiteSpace(recipeBlock))
123:                 return result;
124:
125:             var pattern = new Regex(
126:                 @"\[info\](.*?)\[/info\]",
127:                 RegexOptions.IgnoreCase | RegexOptions.Singleline | RegexOptions.Compiled);
128:             foreach (Match match in pattern.Matches(recipeBlock))
129:             {

131:                 var outputValues = ParseIntList(ExtractTaggedBlock(info, "output"));
132:                 var recipeValues = ParseIntList(ExtractTaggedBlock(info, "recipe"));
133:                 if (outputValues.Count == 0 || recipeValues.Count < 2 || recipeValues.Count % 2 != 0)
134:                     continue;

137:                 if (!entryByOutputId.TryGetValue(outputId, out var catalogEntry))
138:                     continue;

141:                 var pieceCount = recipeValues[1];
142:                 if (pieceId != catalogEntry.EpicPieceId || pieceCount <= 0)
143:                     continue;
144:
145:                 var materials = new List<EpicPieceRecipeEntry>();
146:                 for (var offset = 2; offset + 1 < recipeValues.Count; offset += 2)
147:                 {
148:                     var materialId = recipeValues[offset];
149:                     var materialCount = recipeValues[offset + 1];
150:                     if (materialId <= 0 || materialCount <= 0)
151:                         continue;
152:
153:                     materials.Add(new EpicPieceRecipeEntry
154:                     {
155:                         ItemId = materialId,
156:                         Count = materialCount,
157:                     });
158:                 }
159:
160:                 result[outputId] = new EpicPieceRecipe
161:                 {
162:                     OutputEquipmentId = outputId,
163:                     EpicPieceId = pieceId,
164:                     EpicPieceCount = pieceCount,
165:                     Materials = materials,
166:                 };

172:         private static string ExtractTaggedBlock(string text, string tagName)
173:         {
174:             if (string.IsNullOrEmpty(text) || string.IsNullOrWhiteSpace(tagName))
175:                 return string.Empty;
176:
177:             var open = "[" + tagName + "]";
178:             var close = "[/" + tagName + "]";
179:             var openIndex = text.IndexOf(open, StringComparison.OrdinalIgnoreCase);
180:             if (openIndex < 0)
181:                 return string.Empty;
182:
183:             var start = openIndex + open.Length;
184:             var closeIndex = text.IndexOf(close, start, StringComparison.OrdinalIgnoreCase);
185:             return closeIndex < 0
186:                 ? text.Substring(start)
187:                 : text.Substring(start, closeIndex - start);
188:         }

197:             {

199:                     result.Add(value);
