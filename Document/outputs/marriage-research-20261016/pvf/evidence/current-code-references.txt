501:    /// <summary>
502:    /// Every stride-th value of the objective data, packed into the first three channels.
503:    /// </summary>
504:    private static uint PackChannels(IReadOnlyList<int> intData, int stride)
505:    {
506:        if (intData.Count == 0)
507:            return DefaultInitialTrigger;
508:
509:        var channels = new List<int>();
510:        for (var index = 0; index + stride <= intData.Count; index += stride)
511:            channels.Add(intData[index + stride - 1]);

513:        return channels.Count == 0
514:            ? DefaultInitialTrigger

518:                channels.Count > 2 ? channels[2] : 0);
